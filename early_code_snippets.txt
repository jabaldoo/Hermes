HERMES — early code snippets (first prototype, codename S-MOWE)
================================================================

These are fragments of the very first version of the demonstrator, written on
day one of the hackathon, before the project was renamed HERMES (Hazard Evacuation,
Response and Mesh Embedded System). They are kept here to show where the project
started. None of this code runs in the current version; it has been replaced by the
modules in this repository.

What changed since then, in short:
  - one-file "tick" loop over fixed routes  ->  scenario engine with a mesh network,
    relay stations, store-and-forward, return-to-base and no-fly-zone path planning
  - random weighted statuses  ->  incidents with photos that wait for an operator decision
  - 4 statuses (OK / LUDZIE / ZATOR / POMOC)  ->  8 incident types, emergency-service
    routing, guidance to official KG PSP shelters, compliance statistics
  - blur-only anonymisation  ->  padded regions + downscale/blur so faces stay unreadable

Code comments are in Polish, as in the original files.


----------------------------------------------------------------
1. anonymizer.py (v1) — face anonymisation on board the drone
----------------------------------------------------------------
The core privacy rule was there from the start: the raw frame never leaves the drone.

    # anonymizer.py - anonimizacja twarzy NA POKLADZIE DRONA (edge), zanim cokolwiek opuszcza BSP.
    # Zasada RODO by design: ta funkcja jest jedynym miejscem w systemie, ktore "widzi" surowa klatke
    # z kamera. Do CZK nigdy nie trafia obraz - patrz edge_ai.py i simulator.py.
    import cv2

    _cascade_path = cv2.data.haarcascades + "haarcascade_frontalface_default.xml"
    face_cascade = cv2.CascadeClassifier(_cascade_path)

    TRYBY_DOZWOLONE = ("blur", "pixel", "black")


    def wykryj_twarze(klatka_bgr):
        """Zwraca liste prostokatow (x, y, w, h) wykrytych twarzy na klatce BGR."""
        szara = cv2.cvtColor(klatka_bgr, cv2.COLOR_BGR2GRAY)
        szara = cv2.equalizeHist(szara)
        twarze = face_cascade.detectMultiScale(
            szara, scaleFactor=1.1, minNeighbors=5, minSize=(30, 30)
        )
        return twarze


    def zaanonimizuj(klatka_bgr, twarze, tryb="blur"):
        """Nadpisuje regiony twarzy w klatce (in-place na kopii) i zwraca zmodyfikowana klatke."""
        if tryb not in TRYBY_DOZWOLONE:
            tryb = "blur"
        wynik = klatka_bgr.copy()
        for (x, y, w, h) in twarze:
            roi = wynik[y:y + h, x:x + w]
            if roi.size == 0:
                continue
            if tryb == "blur":
                wynik[y:y + h, x:x + w] = cv2.GaussianBlur(roi, (51, 51), 30)
            elif tryb == "pixel":
                mala = cv2.resize(roi, (8, 8), interpolation=cv2.INTER_LINEAR)
                wynik[y:y + h, x:x + w] = cv2.resize(mala, (w, h), interpolation=cv2.INTER_NEAREST)
            elif tryb == "black":
                wynik[y:y + h, x:x + w] = 0
        return wynik

(Later: a single Gaussian blur on the exact face box turned out to leave faces
recognisable, so the box is now padded and the region is downscaled before blurring.)

3. czk_logic.py (v1) — four statuses and a lookup table
----------------------------------------------------------------
The first "decision support" was a dictionary. Today the operator makes every
decision; the system only recommends and never dispatches anything by itself.

    STATUSY = ("OK", "LUDZIE", "ZATOR", "POMOC")

    # Mapowanie statusu sektora -> jednostki do wyslania + priorytet (1 = najwyzszy)
    _REGULY = {
        "LUDZIE": {"jednostki": ["OSP", "PSP"], "priorytet": 2,
                   "opis": "Wykryto osoby w strefie zagrozenia - wymagana ewakuacja."},
        "ZATOR": {"jednostki": ["Policja"], "priorytet": 3,
                  "opis": "Zator drogowy utrudniajacy ewakuacje/dojazd sluzb."},
        "POMOC": {"jednostki": ["Karetka", "PSP"], "priorytet": 1,
                  "opis": "Osoby poszkodowane / uwiezione - wymagana natychmiastowa pomoc."},
        "OK": {"jednostki": [], "priorytet": 4, "opis": "Brak zagrozenia w sektorze."},
    }


    def rekomenduj_akcje(status_sektora, sektor_id=None, liczba_osob=0):
        """Zwraca rekomendacje dzialania CZK dla danego statusu sektora."""
        regula = _REGULY.get(status_sektora, _REGULY["OK"])
        return {
            "sektor": sektor_id,
            "status": status_sektora,
            "jednostki_rekomendowane": regula["jednostki"],
            "priorytet": regula["priorytet"],
            "opis": regula["opis"],
            "liczba_osob": liczba_osob,
        }

4. simulator.py (v1) — the original swarm "simulation"
----------------------------------------------------------------
Drones walked fixed routes from mock_data.json, and on every tick each one rolled a
weighted die to decide what it "saw". No mesh, no operator, no shelters yet.

    # simulator.py - symulator roju dronow mesh + petla misji edge AI.
    # Zastepuje ROS2/Gazebo prostym generatorem w Pythonie (asyncio), zgodnie z zasada
    # "wszystko co da sie zrobic prosto - robimy prosto".

    # Wagi statusow przy "wykryciu" edge AI w danym sektorze (suma nie musi = 1, losujemy wazone)
    WAGI_STATUSOW = {"OK": 70, "LUDZIE": 14, "ZATOR": 8, "POMOC": 8}
    KROKI_NA_SEGMENT = 6  # ile "tickow" po 1s zajmuje przejscie miedzy dwoma punktami trasy
    UBYTEK_BATERII_NA_TICK = 0.4
    PROG_NISKIEJ_BATERII = 15.0


    def _krok_drona(dron_id):
        """Przesuwa jednego drona o jeden tick po jego trasie, zwraca (lat, lon)."""
        st = _stan_dronow[dron_id]
        trasa = st["trasa"]
        n = len(trasa)
        p1 = trasa[st["seg_idx"] % n]
        p2 = trasa[(st["seg_idx"] + 1) % n]
        t = st["krok"] / KROKI_NA_SEGMENT
        lat, lon = _interpoluj(p1, p2, t)

        st["krok"] += 1
        if st["krok"] >= KROKI_NA_SEGMENT:
            st["krok"] = 0
            st["seg_idx"] = (st["seg_idx"] + 1) % n

        st["bateria"] = max(0.0, st["bateria"] - UBYTEK_BATERII_NA_TICK)
        if st["bateria"] <= PROG_NISKIEJ_BATERII:
            st["status_misji"] = "POWROT_DO_BAZY_LADOWANIE"
            if st["bateria"] <= 0.5:
                st["bateria"] = 100.0
                st["status_misji"] = "PATROL"
        else:
            st["status_misji"] = "PATROL"

        return lat, lon


    async def tick_symulacji(broadcast_fn):
        aktualizacje_dronow = []
        for dron_id, st in _stan_dronow.items():
            lat, lon = _krok_drona(dron_id)
            sektor = _znajdz_sektor(lat, lon)

            # symulacja edge AI: przy kazdym ticku w sektorze losowo (wazone) decydujemy o statusie
            if sektor is not None and random.random() < 0.35:
                nowy_status = _losuj_status_wazony()
                if nowy_status != "OK":
                    liczba_osob = random.randint(1, 12) if nowy_status != "ZATOR" else 0
                    pewnosc = round(random.uniform(0.62, 0.97), 2)
                    db.aktualizuj_status_sektora(sektor["id"], nowy_status)
                    zdarzenie = db.zapisz_zdarzenie(
                        dron_id, sektor["id"], nowy_status, pewnosc, liczba_osob, opis_map[nowy_status]
                    )
            ...
        await broadcast_fn({"typ": "drony", "dane": aktualizacje_dronow})


    async def wywolaj_komunikat_glosowy(dron_id, broadcast_fn):
        """Emisja komunikatu glosowego przez drona 'aktywny' (glosnik) - tylko log, bez realnego audio."""
        tresc = f"Dron {dron_id}: UWAGA, opusc strefe. Uda sie do najblizszego punktu zbiorki."
